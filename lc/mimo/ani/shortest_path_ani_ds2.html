📖 原理简介 · 三种最短路算法对比

三种算法都围绕松弛操作展开：对边 (u, v, w)，若 dist[u] + w < dist[v] 就更新 dist[v]。 区别在于谁来发起松弛、按什么顺序发起。Dijkstra 用优先队列，每次只处理当前最近的点；Bellman-Ford 每轮遍历全部边，做 V−1 轮； SPFA 用普通队列，只把"被更新过的点"重新入队。

算法 1
Dijkstra（堆优化）
贪心：每次取出 dist 最小的未确定节点 u，
用它松弛所有邻居。
复杂度 O((V+E) log V)，要求边权 非负。
算法 2
Bellman-Ford
每轮遍历全部边，全部尝试松弛一次。
重复 V−1 轮，最多能让最短路"边长"增加 V−1。
复杂度 O(VE)，可处理负权，能检测负环。
算法 3
SPFA（队列优化）
只把"刚被更新过的点"重新入队，去松弛它的邻居。
队列空 → 结束。平均 O(kE)，k 一般很小。
最坏 O(VE)，被恶意数据可退化。
对比
核心区别
Dijkstra：一次处理一个确定的点，快但只能非负权
Bellman-Ford：每轮全局扫描，慢但可负权
SPFA：只对"有变化"的点做松弛，兼顾速度与负权
本质
松弛操作 Relax
对边 (u, v, w)：
if dist[u] + w < dist[v]：
　　dist[v] = dist[u] + w；
　　prev[v] = u

🐍 Python 实现 · Dijkstra / Bellman-Ford / SPFA 三种单源最短路 
"""
单源最短路 —— Dijkstra / Bellman-Ford / SPFA。

给定带权图 G = (V, E) 和源点 s，求 s 到每个点的最短距离。

三种算法都围绕"松弛操作"：
    对边 (u, v, w)，若 dist[u] + w < dist[v] 就更新 dist[v]。

区别在于"谁来发起松弛、按什么顺序发起"：
    Dijkstra     —— 每次取 dist 最小的未确定点，非负权 O((V+E) log V)
    Bellman-Ford —— 每轮遍历所有边，V-1 轮，可负权 O(VE)
    SPFA         —— Bellman-Ford 的队列优化，平均 O(kE)
"""

import heapq
from collections import deque


# ============================================================
# 算法 1：Dijkstra（堆优化）—— 非负权，O((V+E) log V)
# ============================================================
def dijkstra(n, adj, src):
    """
    每次从优先队列取出 dist 最小的未确定节点 u，
    用它松弛所有邻居 v。

    关键性质：所有边权非负 → "取出即确定"。
    """
    INF = float('inf')
    dist = [INF] * n
    prev = [-1] * n
    done = [False] * n
    dist[src] = 0

    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if done[u]:
            continue                # 堆里可能有旧版本
        done[u] = True

        for v, w in adj[u]:
            if not done[v] and d + w < dist[v]:
                dist[v] = d + w
                prev[v] = u
                heapq.heappush(pq, (dist[v], v))

    return dist, prev


# ============================================================
# 算法 2：Bellman-Ford —— 可负权，O(VE)
# ============================================================
def bellman_ford(n, edges, src):
    """
    对所有边做 V-1 轮松弛。每轮至少让最短路的"边数"增加 1 层。

    返回 (dist, prev, has_negative_cycle)。
    第 V 轮若还能松弛，说明存在负环。
    """
    INF = float('inf')
    dist = [INF] * n
    prev = [-1] * n
    dist[src] = 0

    for _ in range(n - 1):
        updated = False
        for u, v, w in edges:
            if dist[u] != INF and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                prev[v] = u
                updated = True
        if not updated:
            break                   # 提前结束

    # 第 V 轮：还能松弛 → 存在负环
    has_neg = False
    for u, v, w in edges:
        if dist[u] != INF and dist[u] + w < dist[v]:
            has_neg = True
            break

    return dist, prev, has_neg


# ============================================================
# 算法 3：SPFA —— 队列优化，平均 O(kE)
# ============================================================
def spfa(n, adj, src):
    """
    用队列存"刚被更新过"的节点。
    出队时松弛它的所有邻居，若邻居被更新且不在队列中 → 入队。

    队列空 → 结束。
    """
    INF = float('inf')
    dist = [INF] * n
    prev = [-1] * n
    in_queue = [False] * n
    dist[src] = 0

    q = deque([src])
    in_queue[src] = True

    while q:
        u = q.popleft()
        in_queue[u] = False

        for v, w in adj[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                prev[v] = u
                if not in_queue[v]:
                    q.append(v)
                    in_queue[v] = True

    return dist, prev


# ============================================================
# 路径还原
# ============================================================
def reconstruct_path(prev, src, dst):
    """从 dst 沿 prev 回溯到 src。"""
    path = []
    cur = dst
    while cur != -1:
        path.append(cur)
        if cur == src:
            break
        cur = prev[cur]
    if not path or path[-1] != src:
        return []
    return list(reversed(path))


# ============================================================
# 演示
# ============================================================
if __name__ == '__main__':
    # 6 节点教学图
    # A - B(4)   A - C(2)
    # B - C(5)   B - D(5)
    # C - E(3)   D - E(2)
    # D - F(6)   E - F(7)
    n = 6
    edges = [
        (0, 1, 4), (0, 2, 2),
        (1, 2, 5), (1, 3, 5),
        (2, 4, 3), (3, 4, 2),
        (3, 5, 6), (4, 5, 7),
    ]
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))

    print('=== Dijkstra from A ===')
    d1, p1 = dijkstra(n, adj, 0)
    print('dist =', d1)
    print('A→F 路径 =', reconstruct_path(p1, 0, 5))

    print()
    print('=== Bellman-Ford from A ===')
    d2, p2, neg = bellman_ford(n, edges, 0)
    print('dist =', d2, ' 负环 =', neg)

    print()
    print('=== SPFA from A ===')
    d3, p3 = spfa(n, adj, 0)
    print('dist =', d3)

    assert d1 == d2 == d3, '三种算法应得到相同的 dist'
    print('\n✔ 三种算法结果一致')
