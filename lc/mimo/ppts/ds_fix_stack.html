def is_valid(s: str) -> bool:
    pair = {')': '(', ']': '[', '}': '{'}
    st = []
    for c in s:
        if c in '([{':
            st.append(c)              # 左括号：等着被配对

def evalRPN(tokens):
    st = []
    for t in tokens:
        if t in '+-*/':
            b, a = st.pop(), st.pop()
            if   t == '+': st.append(a + b)
            elif t == '-': st.append(a - b)
            elif t == '*': st.append(a * b)
            else:          st.append(int(a / b))
        else:
            st.append(int(t))
    return st[-1]

队列四动作 · 环形数组

head 与 tail 都只往右走 —— 走到头怎么办？绕回来。
RING BUFFER · capacity = 8
tail = (tail + 1) % capacity，就像钟表指针绕圈。
不取模就会越界，或者白白浪费前面的空位。
HEAD →0TAIL →0N →0
class RingQueue:
    def __init__(self, cap):
        self.a = [None] * cap
        self.head = self.tail = self.n = 0

    def push(self, x):            # enqueue O(1)
        # 满队：(self.n == len(self.a)) 时先判满/扩容
        self.a[self.tail] = x
        self.tail = (self.tail + 1) % len(self.a)
        self.n += 1

    def pop(self):                # dequeue O(1)
        x = self.a[self.head]
        self.head = (self.head + 1) % len(self.a)
        self.n -= 1
        return x

双端队列：两端都能进出

def max_sliding_window(nums, k):
    dq = deque()      # 存下标，值单调递减
    ans = []
    for i, v in enumerate(nums):
        while dq and nums[dq[-1]] <= v:
            dq.pop()  # 又矮又老，永无机会
        dq.append(i)
        if dq[0] <= i - k:
            dq.popleft()      # 滑出窗口
        if i >= k - 1:
            ans.append(nums[dq[0]])
    return ans
WHY · 转变
不维护「窗口里所有数」，只维护「还有可能当最大值的数」。
每个元素只进一次出一次 → 总时间 O(n)。

把「等答案的人」排成一列

栈里存的不是数值，是还没等到答案的下标，从栈底到栈顶单调递减。
RULE · 三条规则
① 新元素 v 来了，看栈顶。
② 只要栈顶 < v：栈顶等到答案 → pop，答案就是 v。
③ 把 v push 进栈，继续等。

为什么 O(n)？每个下标只进一次、只出一次，总操作 ≤ 2n。
单调栈：n = 10^5 → 2×10^5 次操作
ANS →[4, 2, 4, -1, -1]STACK →[3, 4]
def next_greater(nums):
    n = len(nums)
    ans = [-1] * n
    st = []                    # 存下标，值单调递减
    for i in range(n):
        while st and nums[st[-1]] < nums[i]:
            ans[st.pop()] = nums[i]   # i 就是它等的答案
        st.append(i)
    return ans
KEY · 一句话
栈里永远是「还没等到答案，而且越来越没希望」的人。

def largest_area(h):
    h = h + [0]                # 哨兵：逼出栈里剩下的
    st, best = [], 0
    for i, v in enumerate(h):
        while st and h[st[-1]] > v:
            height = h[st.pop()]
            left = st[-1] if st else -1
            best = max(best, height * (i - left - 1))
        st.append(i)
    return best

BFS 为什么天然最短

它是按层扩张的：第 k 层的格子，一定是被第 k−1 层第一次到达的。
LAYERS · 每一环 = 距离为 d 的所有格子
ANS →4STRUCT →deque
def shortest(grid):
    n = len(grid)
    if grid[0][0] == 1: return -1
    q = deque([( 0, 0, 1 )])   # (x, y, 步数)
    seen = {(0, 0)}
    while q:
        x, y, d = q.popleft()
        if x == y == n - 1: return d
        for dx in (-1,0,1):
            for dy in (-1,0,1):
                nx, ny = x + dx, y + dy
                if 0 <= nx < n and (nx,ny) not in seen:
                    seen.add((nx, ny))
                    q.append((nx, ny, d + 1))
    return -1

陷阱二：递归栈溢出

Python 的调用栈不是无限的，默认约 1000 层。

FIX · 两条路

sys.setrecursionlimit(100000)  # 只是把栈做大
# 更稳：递归 → 显式栈
st = [start]
while st:
    x = st.pop()
    for y in next_of(x):
        st.append(y)
COMPARE · 递归 vs 显式栈
写法栈在哪风险
递归系统调用栈深度受限、慢
显式栈堆内存可控、不炸
KEY · 一句话
递归 = 用系统栈做 DFS。写成显式栈，你就能自己决定「谁先出去」—— 又回到今天这根管子。

陷阱三：结构选错

用 list 当队列、用 deque 当栈 —— 都能跑，但代价不同。
✗ LIST 当队列
q = [1, 2, 3]
x = q.pop(0)  # O(n)
每次出队，后面全体前移一格。10 万次 → 退化到 10^10 量级。
✓ DEQUE 当队列
q = deque([1, 2, 3])
x = q.popleft()  # O(1)
两头都是 O(1)。若只用 append / pop，用 list 就够了。
需求该用别用原因
栈list.append / list.popdeque.poplist 更简单，缓存更友好
队列deque.append / popleftlist.pop(0)后者是 O(n)
优先队列heapq每次 sorted排序 O(n log n)，堆只需 O(log n)
滑动窗口最值单调 deque每窗口重扫后者 O(nk)

append / popleft　O(1)
排队、BFS、层序、缓冲、最短路。
CARD 3 · 双端 / 堆

淘汰「没机会」的，留下候选者。
