两数之和

Python
def two_sum(nums, target):
    seen = {}
    for i, x in enumerate(nums):
        if target - x in seen:
            return [seen[target-x], i]
        seen[x] = i
C++
vector<int> twoSum(vector<int>& nums, int target) {
    unordered_map<int,int> seen;
    for (int i = 0; i < nums.size(); i++) {
        int need = target - nums[i];
        if (seen.count(need)) return {seen[need], i};
        seen[nums[i]] = i;
    }
    return {};
}

综合示例：BFS 最短路

上篇收尾
Python py
from collections import deque

def bfs(start, target, graph):
    q = deque([(start, 0)])
    visited = {start}
    while q:
        u, d = q.popleft()
        if u == target: return d
        for v in graph[u]:
            if v not in visited:
                visited.add(v)
                q.append((v, d+1))
    return -1
C++ cpp
int bfs(int start, int target, vector<vector<int>>& graph) {
    queue<pair<int,int>> q;
    vector<bool> visited(graph.size(), false);
    q.push({start, 0});
    visited[start] = true;
    while (!q.empty()) {
        auto [u, d] = q.front(); q.pop();
        if (u == target) return d;
        for (int v : graph[u]) {
            if (!visited[v]) {
                visited[v] = true;
                q.push({v, d+1});
            }
        }
    }
    return -1;
}

数据交换的不同

swap 语义
Python py
# 一行搞定（元组打包/解包）
a, b = b, a

# 容器交换引用 O(1)
arr1, arr2 = arr2, arr1

# 字符串交换
s1, s2 = s2, s1
C++ cpp
// ① 临时变量（最常用）
int t = a; a = b; b = t;

// ② std::swap（推荐）
swap(a, b);

// ③ C++11 起也可用 tuple 解包
tie(a, b) = make_tuple(b, a);

// 容器交换 O(1)（交换内部指针）
vector<int> v1 = {1,2,3}, v2 = {4,5};
swap(v1, v2);   // 高效

// 原生数组 swap 是 O(n) 拷贝！
int a[100], b[100];

差异： Python 交换容器永远是 O(1)（换名字）；C++ 的 swap 对 vector/string/map 是 O(1)，对原生数组是 O(n)。
